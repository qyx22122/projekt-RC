\section{Analiza podatkov}
\begin{frame}
\frametitle{Analiza podatkov}
	\begin{figure}
	\centering
	\includegraphics[height = .7\textheight]{potisk}
	\end{figure}

	\begin{equation}
		F \Delta t = I = \SI{17,3}{Ns} \pm \SI{1,6}{Ns}\nonumber
	\end{equation}
\end{frame}
\begin{frame}
\frametitle{Povprečni potisk in specifični impulz}

\begin{align}
\onslide<1->{
\bar{F} &= \frac{I}{t}
= \frac{\SI{17,3}{Ns} \pm \SI{1,6}{Ns}}
       {\SI{3,6}{s} \pm \SI{0,1}{s}}
= \SI{4,8}{N} \pm \SI{0,6}{N}
} \nonumber \\[1em]
\onslide<2->{
I_{SP} &= \frac{I}{m \cdot g_0}
= \frac{\SI{17,3}{Ns} \pm \SI{1,6}{Ns}}
       {(\SI{0,075}{kg} \pm \SI{0,001}{kg}) \cdot \SI{9.80665}{m/s^2}}
= \SI{23,5}{s} \pm \SI{2,3}{s}
} \nonumber
\end{align}
\end{frame}

\begin{frame}
\begin{align}
\frametitle{Izhodna hitrost izpuhov in sprememba hitrost rakete}
\onslide<1->{
\bar{v_e} &= \frac{I}{m}
= \frac{\SI{17,3}{Ns} \pm \SI{1,6}{Ns}}
       {\SI{0,075}{kg} \pm \SI{0,001}{kg}}
= \SI{230}{m/s} \pm \SI{23}{m/s}
} \nonumber \\[1em]
\onslide<2->{
\Delta v &= v_e \ln\!\left(\frac{m_0}{m_1}\right) = (\SI{230}{m/s} \pm \SI{23}{m/s})
   \cdot
   \ln\!\left(
   \frac{\SI{0,170}{kg} \pm \SI{0,001}{kg}}
        {\SI{0,095}{kg} \pm \SI{0,001}{kg}}
   \right) \nonumber \\[1em]
	&= \SI{134}{m/s} \pm \SI{17}{m/s}
} \nonumber
\end{align}
\end{frame}
