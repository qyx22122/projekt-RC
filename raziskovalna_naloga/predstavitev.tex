\onslide<2->{
I_{SP} &= \frac{I}{m \cdot g_0}
= \frac{\SI{17,26}{Ns} \pm \SI{1,55}{Ns}}
       {(\SI{0,075}{kg} \pm \SI{0,001}{kg}) \cdot \SI{9.80665}{m/s^2}}

} \nonumber \\[1em]

\bar{v_e} &= \frac{I}{m}
= \frac{\SI{17,26}{Ns} \pm \SI{1,55}{Ns}}
       {\SI{0,075}{kg} \pm \SI{0,001}{kg}}
= \SI{230}{m/s} \pm \SI{23}{m/s}
} \nonumber \\[1em]
\onslide<4->{
\Delta v &= v_e \ln\!\left(\frac{m_0}{m_1}\right) = (\SI{230}{m/s} \pm \SI{23}{m/s})

} \nonumber
