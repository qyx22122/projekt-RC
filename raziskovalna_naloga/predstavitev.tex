\section{Analiza podatkov}
\begin{frame}
\frametitle{Analiza podatkov}
	\begin{figure}
	\centering
	\includegraphics[height = .5\textheight]{potisk}
	\end{figure}

	\begin{equation}
		I = \SI{17,26}{Ns} \pm \SI{1,55}{Ns}\nonumber
	\end{equation}
\end{frame}
\begin{frame}
	
	\begin{align}
		\bar{F} &= \frac{I}{t} = \frac{\SI{17,26}{Ns} \pm \SI{1,55}{Ns}}{\SI{3,6}{s} \pm \SI{0,1}{s}} = \SI{4,8}{N} \pm \SI{0,6}{N} \nonumber \\[1em]
		I_{SP} &= \frac{I}{m \vdot g_0} = \frac{\SI{17,26}{Ns} \pm \SI{1,55}{Ns}}{(\SI{0,075}{kg} \pm \SI{0,001}) \vdot \SI{9.80665}{m/s^2}} = \SI{23,5}{s} \pm \SI{2,3}{s} \nonumber \\[1em]
		\bar{v_e} &= \frac{I}{m} = \frac{\SI{17,26}{Ns} \pm \SI{1,55}{Ns}}{\SI{0,075}{kg} \pm \SI{0,001}{kg}} = \SI{230}{m/s} \pm \SI{23}{m/s} \nonumber \\[1em]
		\Delta v &= v_e \ln{\frac{m_0}{m_1}} = (\SI{230}{m/s} \pm \SI{23}{m/s})\vdot \ln{\frac{\SI{0,170}{kg} \pm \SI{0,001}{kg}}{\SI{0,095}{kg} \pm \SI{0,001}{kg}}} = \nonumber \\[1em]
		&= \SI{134}{m/s} \pm \SI{17}{m/s} \nonumber
	\end{align}
\end{frame}
