\begin{frame}
\frametitle{Specifičnen impulz}
	\begin{columns}
		\column{.6\textwidth}
			\begin{equation}
				I = G_1 - G_0 = \int F dt
			\end{equation}
			\begin{equation}\label{eq:ISP}
				I_{SP} = \frac{I}{m \vdot g_0}
			\end{equation}
			\begin{equation} \label{eq:ISP_v}
				I_{SP} = \frac{v_e}{g_0}
			\end{equation}
			Enačba Ciolkovskega:
			\begin{equation} \label{eq:Ciolkovskega}
				\Delta v = v_e \ln{\frac{m_0}{m_1}} = I_{SP} g_0 \ln{\frac{m_0}{m_1}}
			\end{equation}
